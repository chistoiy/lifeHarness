/* 分类 / 单位自定义（同一套管理器） */
let hbCatF='全部',hbMgrKind='cat';
function hbMgrCfg(k){return k==='unit'
 ?{title:'记录量单位',sub:'改名即时同步到习惯与组内子项 · 删除前需先安置使用者',
   list:()=>HB_UNITS,set:(i,v)=>HB_UNITS[i]=v,splice:i=>HB_UNITS.splice(i,1),push:v=>HB_UNITS.push(v),
   used:u=>HABITS.filter(h=>h.unit===u).length+HABITS.reduce((a,h)=>a+(h.subs||[]).filter(s=>s.unit===u).length,0),
   ren:(o,n)=>{HABITS.forEach(h=>{if(h.unit===o)h.unit=n});HABITS.forEach(h=>(h.subs||[]).forEach(s=>{if(s.unit===o)s.unit=n}));if(heUnit===o)heUnit=n},
   mv:(c,to)=>{HABITS.forEach(h=>{if(h.unit===c)h.unit=to});HABITS.forEach(h=>(h.subs||[]).forEach(s=>{if(s.unit===c)s.unit=to}));if(heUnit===c)heUnit=to},
   goto:null}
 :{title:'习惯分类',sub:'改名即时同步到已归属习惯 · 删除前需先安置其下习惯',
   list:()=>HB_CATS,set:(i,v)=>HB_CATS[i]=v,splice:i=>HB_CATS.splice(i,1),push:v=>HB_CATS.push(v),
   used:c=>HABITS.filter(h=>h.cat===c).length,
   ren:(o,n)=>{HABITS.forEach(h=>{if(h.cat===o)h.cat=n});if(heCat===o)heCat=n;if(hbCatF===o)hbCatF=n},
   mv:(c,to)=>{HABITS.forEach(h=>{if(h.cat===c)h.cat=to});if(heCat===c)heCat=to;if(hbCatF===c)hbCatF=to},
   goto:c=>{hbCatPick(c);closeSheet('sheetHabitMgr')}}}
function hbCatRow(){const el=document.getElementById('hbCatRow');if(!el)return;
 const book=hbBook();
 el.innerHTML=['全部'].concat(HB_CATS).map(c=>{const n=c==='全部'?book.length:book.filter(h=>h.cat===c).length;
   return `<button class="chip whitespace-nowrap !text-[10px] ${hbCatF===c?'chip-on':''}" onclick="hbCatPick('${c}')">${c} ${n}</button>`}).join('')+
  `<button class="chip whitespace-nowrap !text-[10px]" onclick="hbMgrOpen('cat')">${hbi('cog','w-[11px] h-[11px] inline -mt-0.5')} 分类管理</button>`}
function hbCatPick(c){hbCatF=c;renderHabits()}
function hbMgrOpen(k){hbMgrKind=k||hbMgrKind;const cfg=hbMgrCfg(hbMgrKind),el=document.getElementById('hmList'),ls=cfg.list();
 const tt=document.getElementById('hmTitle');if(tt)tt.textContent=cfg.title;
 const sb=document.getElementById('hmSub');if(sb)sb.textContent=cfg.sub;
 if(el)el.innerHTML=ls.map((c,i)=>{const n=cfg.used(c);
  return `<div class="card !py-2.5 !px-3">
   <div class="flex items-center gap-1.5">
    <input class="inp flex-1 !py-1.5 text-[12.5px]" value="${c}" onchange="hbMgrRen(${i},this.value)">
    <button class="w-7 h-7 rounded-lg bg-slate-100 dark:bg-slate-800 text-slate-500 flex items-center justify-center shrink-0${i?'':' opacity-30'}" onclick="hbMgrMove(${i},-1)" title="上移">${hbi('up','w-[13px] h-[13px]')}</button>
    <button class="w-7 h-7 rounded-lg bg-slate-100 dark:bg-slate-800 text-slate-500 flex items-center justify-center shrink-0${i===ls.length-1?' opacity-30':''}" onclick="hbMgrMove(${i},1)" title="下移">${hbi('down','w-[13px] h-[13px]')}</button>
    <button class="w-7 h-7 rounded-lg bg-rose-50 dark:bg-rose-500/15 text-rose-500 flex items-center justify-center shrink-0" onclick="hbMgrDel(${i})" title="删除">${hbi('trash','w-[13px] h-[13px]')}</button></div>
   <div class="mt-1 flex items-center gap-2 text-[10px] text-slate-400">
    <span>${n?(cfg.goto?'<button class="link-act" onclick="hbMgrGoto('+i+')">下属 '+n+' 项 · 查看</button>':'被 '+n+' 处使用'):'暂无使用'}</span>
    ${n?'<button class="link-act ml-auto" onclick="hbMgrPanel('+i+')">迁移后删除</button>':''}</div>
   <div id="hmMv'+i+'" class="hidden mt-2 items-center gap-2 text-[11px]">
    <select id="hmMvS'+i+'" class="sel flex-1 !text-[11px]">'+ls.filter((x,k)=>k!==i).map(x=>'<option>'+x+'</option>').join('')+'</select>
    <button class="mini-btn !text-[10px]" onclick="hbMgrMerge('+i+')">迁移并删除</button></div>
  </div>`}).join('')||'<div class="card text-[11px] text-slate-400 text-center !py-6">已清空 · 添加一个开始归类</div>';
 const nw=document.getElementById('hmNew');if(nw)nw.placeholder='新'+(hbMgrKind==='unit'?'单位':'分类')+'名称';
 if(nw)nw.value='';openSheet('sheetHabitMgr')}
function hbMgrGoto(i){const cfg=hbMgrCfg(hbMgrKind);if(cfg.goto)cfg.goto(cfg.list()[i])}
function hbMgrPanel(i){const el=document.getElementById('hmMv'+i);if(!el)return;
 const on=el.classList.contains('flex');el.classList.toggle('hidden',on);el.classList.toggle('flex',!on)}
function hbMgrAdd(){const cfg=hbMgrCfg(hbMgrKind),el=document.getElementById('hmNew'),v=(el.value||'').trim();
 if(!v){toast('先输入名称');return}
 if(cfg.list().indexOf(v)>=0){toast('已有同名'+(hbMgrKind==='unit'?'单位':'分类'));return}
 cfg.push(v);hbMgrSync();toast('已添加「'+v+'」')}
function hbMgrRen(i,v){const cfg=hbMgrCfg(hbMgrKind),ls=cfg.list(),old=ls[i],nv=(v||'').trim();
 if(!nv||nv===old){hbMgrSync();return}
 if(ls.indexOf(nv)>=0){toast('已有同名 · 未改名');hbMgrSync();return}
 cfg.set(i,nv);cfg.ren(old,nv);hbMgrSync();
 toast('「'+old+'」已改为「'+nv+'」· 同步 '+cfg.used(nv)+' 处')}
function hbMgrMove(i,d){const cfg=hbMgrCfg(hbMgrKind),ls=cfg.list(),j=i+d;
 if(j<0||j>=ls.length)return;const t=ls[i];ls[i]=ls[j];ls[j]=t;hbMgrSync()}
function hbMgrDel(i){const cfg=hbMgrCfg(hbMgrKind),ls=cfg.list(),c=ls[i];
 if(ls.length<=1){toast('至少保留一个'+(hbMgrKind==='unit'?'单位':'分类'));return}
 if(cfg.used(c)){hbMgrPanel(i);toast('「'+c+'」仍被使用 · 选择去向后可删除');return}
 cfg.splice(i);hbMgrSync();toast('已删除「'+c+'」')}
function hbMgrMerge(i){const cfg=hbMgrCfg(hbMgrKind),ls=cfg.list(),c=ls[i],to=(document.getElementById('hmMvS'+i)||{}).value;
 if(!to||to===c){toast('请选择迁移目标');return}
 const n=cfg.used(c);cfg.mv(c,to);cfg.splice(i);hbMgrSync();
 toast('已把 '+n+' 处迁到「'+to+'」· 并删除「'+c+'」')}
function hbMgrSync(){const s=document.getElementById('sheetHabitMgr');
 renderHabits();renderHabitStat();
 if(typeof heRender==='function'&&!document.getElementById('sheetHabitEdit').classList.contains('hidden'))heRender();
 if(s&&!s.classList.contains('hidden')){const el=document.getElementById('hmList'),keep=el?el.scrollTop:0;hbMgrOpen();if(el)el.scrollTop=keep}}
