function hbAgg(dim,off){const book=hbBook();let cnt=0,val=0,rec=0;const days=hbDimDays(dim);
 book.forEach(h=>{const t=hbTot(h,dim,off);cnt+=t.cnt;val+=t.val;rec+=hbRecN(h,dim,off)});
 return {cnt:cnt,val:val,rec:rec,days:days,n:book.length,rate:Math.min(100,Math.round(cnt/Math.max(1,book.length*days)*100))}}
function hbDlt(cur,prev,betterUp){if(prev==null)return '<span class="text-slate-400">无数据</span>';
 const d=cur-prev,p=prev?Math.round(d/prev*100):(d?100:0);
 if(!p)return '<span class="text-slate-400">持平</span>';
 const good=betterUp?d>0:d<0;
 return '<span class="'+(good?'text-emerald-500':'text-rose-500')+' font-bold">'+(d>0?'+':'')+p+'%</span>'}
