/* 打卡设置 */
const HB_SET={spanFromStart:true,autoChk:true,remindAll:true,weekStart:0,rateBase:0};
const HB_SET_ITEMS=[
 ['spanFromStart','日均按开始跟踪日起算','统计里的日均与达成率只计入习惯开始之后的天数 · 年中新增不会被全年摊薄','sw'],
 ['autoChk','自动达成源直接打卡','番茄 / 阅读计时 / 日程满足条件即记为已打卡，流水标注「自动」','sw'],
 ['remindAll','允许提醒','关闭后所有习惯提醒静默，打卡与统计照常','sw'],
 ['rateBase','达成率口径','按应打卡天数或按周期总天数计算达成率','seg',['应打卡天数','周期天数']],
 ['weekStart','周起始日','影响日历与周统计的分列方式','seg',['周一','周日']]];
function hbSetOpen(){hbSetRender();openSheet('sheetHabitSet')}
function hbSetRender(){const el=document.getElementById('hbSetList');if(!el)return;
 el.innerHTML=HB_SET_ITEMS.map(it=>{const k=it[0],t=it[1],d=it[2],tp=it[3],v=HB_SET[k];
  if(tp==='seg')return `<div class="card !py-2.5 !px-3"><b class="text-[12.5px]">${t}</b><div class="mt-1 text-[10px] text-slate-400 leading-4">${d}</div>
    <div class="mt-2 flex gap-1.5">${it[4].map((s,i)=>`<button class="chip flex-1 justify-center !text-[10.5px] ${v===i?'chip-on':''}" onclick="hbSetSeg('${k}',${i})">${s}</button>`).join('')}</div></div>`;
  return `<div class="card !py-2.5 !px-3"><div class="flex items-center gap-2"><div class="flex-1 min-w-0"><b class="text-[12.5px] block">${t}</b>
    <span class="text-[10px] text-slate-400 leading-4 block mt-0.5">${d}</span></div><span class="toggle${v?' on':''}" onclick="hbSetT('${k}')"></span></div></div>`}).join('')}
function hbSetT(k){HB_SET[k]=!HB_SET[k];hbSetRender();hbSetAfter(k)}
function hbSetSeg(k,i){HB_SET[k]=i;hbSetRender();hbSetAfter(k)}
function hbSetAfter(k){renderHabits();renderHabitStat();if(curPage==='p-habitDetail')renderHabitDetail();
 toast('已更新设置 · '+HB_SET_ITEMS.find(x=>x[0]===k)[1])}
function hbDueDays(h){if(HB_SET.rateBase)return 7;
 return h.cycle==='month'?(h.targetM||1):h.cycle==='week'?(h.days&&h.days.length?h.days.length:1):7}
function hbSpanD(h){const s=h.start?new Date(h.start):new Date(2026,8,19);
 return Math.max(1,Math.round((new Date(2026,8,19)-new Date(s.getFullYear(),s.getMonth(),s.getDate()))/864e5)+1)}
function hbPeriodStart(dim,off){const b=new Date(2026,8,19),back=off*hbDimDays(dim);
 if(dim==='a')return new Date(b.getFullYear()-off,0,1);
 if(dim==='m'){const t=new Date(b.getFullYear(),b.getMonth()-off,1);return t}
 return new Date(b.getFullYear(),b.getMonth(),b.getDate()-6-back)}
function hbEffDays(h,dim,off){const full=hbDimDays(dim);
 if(!HB_SET.spanFromStart||!h.start)return full;
 const s=new Date(h.start),ps=hbPeriodStart(dim,off);
 if(s>ps)return Math.max(1,Math.round((new Date(2026,8,19)-s)/864e5)+1);
 return full}
function hbWdHead(){return HB_SET.weekStart?['日','一','二','三','四','五','六']:['一','二','三','四','五','六','日']}
