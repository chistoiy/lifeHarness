/* 宽限卡 */
function hbGl(h){if(h.gl==null)h.gl=HB_SET.grace||0;return h.gl}
function hbFreeze(i){const h=HABITS[i];if(hbGl(h)<=0){toast('本月宽限次数已用完');return}
 if(h.w[6]){toast('今天已打卡 · 无需宽限');return}
 h.gl=hbGl(h)-1;h.fz=h.fz||[0,0,0,0,0,0,0];h.fz[6]=1;h.w[6]=1;h.mDone=(h.mDone||0)+1;
 h.nt=h.nt||{};h.nt[6]='使用宽限卡';
 renderHabits();renderHabitStat();if(curPage==='p-habitDetail')renderHabitDetail();
 toast('已用一张宽限 · 连续 '+h.streak+' 天未中断 · 本月剩 '+h.gl+' 张')}

/* 自动达成待确认 */
function hbPendOpen(){hbPendRender();openSheet('sheetHabitPend')}
function hbPendRender(){const el=document.getElementById('hpList');if(!el)return;
 const book=hbBook().filter(h=>(h.pend||[]).length);
 el.innerHTML=book.length?book.map(h=>`<div class="card !py-2.5 !px-3">
   <div class="flex items-center gap-2"><span class="w-7 h-7 rounded-lg bg-brand-50 dark:bg-brand-500/15 text-brand-500 flex items-center justify-center shrink-0">${hbi(h.ic,'w-[14px] h-[14px]')}</span>
    <div class="flex-1 min-w-0"><b class="text-[12.5px] block truncate">${h.n}</b>
     <span class="text-[10px] text-slate-400">${h.pend.map(p=>p.src+' · '+p.t+(p.val?' · 量 '+p.val:'')).join(' / ')}</span></div></div>
   <div class="mt-2 flex gap-2"><button class="mini-btn flex-1" onclick="hbPendOk('${h.id}')">确认打卡</button>
    <button class="mini-btn flex-1" onclick="hbPendNo('${h.id}')">忽略</button></div></div>`).join('')
  :'<div class="card text-[11px] text-slate-400 text-center !py-8">没有待确认的自动打卡</div>';
 const b=document.getElementById('hpAll');if(b)b.style.display=book.length?'':'none'}
function hbPendOk(id){const h=HABITS.find(x=>x.id===id);if(!h||!h.pend)return;
 const p=h.pend[0];h.pend.shift();h.w[6]=1;h.at=h.at||['','','','','','',''];h.at[6]=p.t;
 h.streak=(h.streak||0)+1;h.mDone=(h.mDone||0)+1;if(h.streak>h.best)h.best=h.streak;
 if(p.val){if(h.group){h.sv=h.sv||[[],[],[],[],[],[],[]];h.sv[6]=h.subs.map(()=>Math.round(p.val/h.subs.length));h.v[6]=p.val}
  else{h.v=h.v||[0,0,0,0,0,0,0];h.v[6]=p.val}}
 hbPendRender();renderHabits();renderHabitStat();toast('已确认「'+h.n+'」· 来源 '+p.src)}
function hbPendNo(id){const h=HABITS.find(x=>x.id===id);if(!h||!h.pend)return;
 const p=h.pend.shift();hbPendRender();renderHabits();toast('已忽略 · 来源 '+p.src+' 未计入打卡')}
function hbPendAll(){const list=hbBook().filter(h=>(h.pend||[]).length).map(h=>h.id);
 if(!list.length){toast('没有待确认项');return}
 list.forEach(id=>{while((HABITS.find(x=>x.id===id).pend||[]).length)hbPendOk(id)});
 hbPendRender();toast('已确认 '+list.length+' 个习惯的自动打卡')}
function hbPendSim(){const h=hbBook().find(x=>(x.src||[]).some(s=>s!=='手动打卡'));
 if(!h){toast('没有绑定自动达成源的习惯');return}
 const src=(h.src||[]).find(s=>s!=='手动打卡');
 syncHabit('打卡：'+h.n+'（'+src+'）');
 closeSheet('sheetHabitSrc');toast('已模拟一次「'+src+'」达成')}
