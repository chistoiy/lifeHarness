/* 月历（补卡 / 详情近 30 天共用） */
function hbBase(){return new Date(2026,8,19)}
function hbDateOf(back){const d=new Date(hbBase().getTime());d.setDate(d.getDate()-back);return d}
function hbWdIdx(d){return (d.getDay()+6)%7}
function hbMd(d){return (d.getMonth()+1)+'-'+String(d.getDate()).padStart(2,'0')}
function hbBackOf(d){return Math.round((hbBase()-new Date(d.getFullYear(),d.getMonth(),d.getDate()))/864e5)}
function hbDayGet(h,back){if(back>=0&&back<=6){const k=6-back;return {ok:!!h.w[k],val:h.group?((h.sv&&h.sv[k])||[]):(((h.v&&+h.v[k])||0)),at:(h.at&&h.at[k])||'',bf:!!(h.bf&&h.bf[k])}}
 const e=(h.ex||{})[back]||{};return {ok:!!e.ok,val:h.group?(e.sv||[]):((+e.v)||0),at:e.at||'',bf:!!e.bf}}
function hbDaySet(h,back,ok,v,meta){meta=meta||{};
 if(back>=0&&back<=6){const k=6-back;h.w[k]=ok?1:0;
   if(v!=null){if(h.group){h.sv=h.sv||[[],[],[],[],[],[],[]];h.sv[k]=v.slice();h.v[k]=v.reduce((a,x)=>a+(+x||0),0)}
    else{h.v=h.v||[0,0,0,0,0,0,0];h.v[k]=+v||0}}
   if(meta.at!=null){h.at=h.at||['','','','','','',''];h.at[k]=meta.at}
   if(meta.bf){h.bf=h.bf||[0,0,0,0,0,0,0];h.bf[k]=1}}
  else{h.ex=h.ex||{};const e=h.ex[back]||(h.ex[back]={ok:0});e.ok=ok?1:0;
   if(v!=null){if(h.group){e.sv=v.slice();e.v=v.reduce((a,x)=>a+(+x||0),0)}else e.v=+v||0}
   if(meta.at!=null)e.at=meta.at;if(meta.bf)e.bf=1}}
function hbValAt(h,back){const v=hbDayGet(h,back).val;return h.group?(v||[]).reduce((a,x)=>a+(+x||0),0):(+v||0)}
function hbDayTip(h,back){const g=hbDayGet(h,back),d=hbDateOf(back);
 return hbMd(d)+' 周'+'日一二三四五六'[d.getDay()]+' · '+(g.ok?(h.type==='pos'?'已打卡':'未触发'):'未记录')
  +(g.at?' '+g.at:'')+(hbValAt(h,back)?' · 数值 '+hbValAt(h,back):'')+(g.bf?' · 补记':'')}
/* monthOff：0 本月，-1 上月；返回 7 列格子（null 为空白） */
function hbMonthCells(monthOff){const b=hbBase(),t=new Date(b.getFullYear(),b.getMonth()+(monthOff||0),1);
 const y=t.getFullYear(),m=t.getMonth(),last=new Date(y,m+1,0).getDate(),lead=hbWdIdx(new Date(y,m,1));
 const cells=[];for(let k=0;k<lead;k++)cells.push(null);
 for(let d=1;d<=last;d++)cells.push({y:y,m:m,d:d});
 return cells}
function hbMonthHtml(h,cells,sel,tap){const b=hbBase(),lim=new Date(b.getFullYear(),b.getMonth(),b.getDate());
 return cells.map(c=>{if(!c)return '<span></span>';
  const dt=new Date(c.y,c.m,c.d);if(dt>lim)return '<span></span>';
  const back=hbBackOf(dt),g=hbDayGet(h,back),val=hbValAt(h,back),isToday=back===0,on=sel&&sel.indexOf(back)>=0;
  const bg=on?'bg-brand-50 dark:bg-brand-500/15 ring-2 ring-brand-500':(g.ok?(h.type==='pos'?'bg-emerald-50 dark:bg-emerald-500/15':'bg-teal-50 dark:bg-teal-500/15'):'bg-slate-100 dark:bg-slate-800');
  const fg=isToday?'text-brand-500':(g.ok?(h.type==='pos'?'text-emerald-600 dark:text-emerald-400':'text-teal-600 dark:text-teal-400'):'text-slate-400');
  return `<button class="h-[38px] rounded-[10px] flex flex-col items-center justify-center ${bg}" ${tap?`onclick="${tap}(${back})"`:''} data-b="${back}" title="${hbDayTip(h,back)}">
   <span class="text-[10.5px] font-bold leading-none ${fg}">${c.d}</span>
   <span class="flex gap-0.5 mt-1 h-1 items-center">${g.ok?'<i class="w-1 h-1 rounded-full '+(h.type==='pos'?'bg-emerald-500':'bg-teal-500')+'"></i>':''}${val?'<i class="w-1 h-1 rounded-full bg-brand-500"></i>':''}</span></button>`}).join('')}
function hbMonthLabel(monthOff){const b=hbBase(),t=new Date(b.getFullYear(),b.getMonth()+(monthOff||0),1);
 return t.getFullYear()+' 年 '+(t.getMonth()+1)+' 月'}
