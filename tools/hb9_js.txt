/* 补卡 */
let bfH=null,bfSel=[],bfV=[],bfMonth=0,bfVk=null;
function bfHabit(){return HABITS.find(x=>x.id===bfH)||hbBook()[0]||HABITS[0]}
function hbBackfillOpen(id){if(id)bfH=id;const h=bfHabit();if(!h){toast('还没有可补的习惯');return}
 bfH=h.id;bfSel=[];bfMonth=0;bfVk=null;bfRender()}
function bfPick(id){bfH=id;bfSel=[];bfMonth=0;bfVk=null;bfRender()}
function bfMonthStep(d){bfMonth=Math.max(-11,Math.min(0,bfMonth+d));bfRender()}
function bfTap(back){bfSel=bfSel.indexOf(back)>=0?bfSel.filter(x=>x!==back):bfSel.concat([back]);bfRender()}
function bfClear(){if(!bfSel.length){toast('当前没有选中日期');return}bfSel=[];bfRender()}
function bfValInit(h){bfV=h.group?h.subs.map(s=>+s.goal||0):[+h.goal||0];bfVk=h.id}
function bfValRows(h){const el=document.getElementById('bfValRows');if(!el)return;
 el.innerHTML=h.group?h.subs.map((s,k)=>`<div class="flex items-center gap-2 text-[11.5px]">
    <span class="w-16 shrink-0 truncate">${s.n}</span>
    <input class="inp !w-16 !py-1 text-center" type="number" min="0" value="${bfV[k]||0}" onchange="bfSet(${k},this.value)">
    <span class="text-[10px] text-slate-400 w-8">${s.unit}</span>
    <span class="ml-auto text-[9.5px] text-slate-400">目标 ${s.goal}</span></div>`).join('')
  :`<div class="flex items-center gap-2 text-[11.5px]"><span class="w-16 shrink-0 truncate">数值</span>
    <input class="inp !w-16 !py-1 text-center" type="number" min="0" value="${bfV[0]||0}" onchange="bfSet(0,this.value)">
    <span class="text-[10px] text-slate-400">${h.unit||'次'}</span>
    <span class="ml-auto text-[9.5px] text-slate-400">目标 ${h.goal||0}</span></div>`}
function bfSet(k,v){bfV[k]=Math.max(0,+v||0);bfN()}
function bfN(){const el=document.getElementById('bfN');if(el)el.textContent='已选 '+bfSel.length+' 天';
 const t=document.getElementById('bfValTip');if(t)t.textContent=bfSel.length?('将补记到所选 '+bfSel.length+' 天'):'未选日期';
 const nm=document.getElementById('bfNextM');if(nm)nm.className='w-7 h-7 rounded-lg bg-slate-100 dark:bg-slate-800 text-slate-500 flex items-center justify-center shrink-0'+(bfMonth>=0?' opacity-30':'')}
function bfRender(){const h=bfHabit();if(!h)return;
 document.getElementById('bfTitle').textContent='补卡 · '+h.n;
 document.getElementById('bfHabits').innerHTML=HABITS.filter(x=>!x.arch).map(x=>`<button class="chip whitespace-nowrap !text-[10px] ${x.id===h.id?'chip-on':''}" onclick="bfPick('${x.id}')">${x.n}</button>`).join('')||'<span class="text-[10px] text-slate-400">暂无可补习惯</span>';
 document.getElementById('bfHead').innerHTML=['一','二','三','四','五','六','日'].map(d=>`<span>${d}</span>`).join('');
 document.getElementById('bfMonthLbl').textContent=hbMonthLabel(bfMonth)+(bfMonth===0?' · 今天所在月':'');
 document.getElementById('bfGrid').innerHTML=hbMonthHtml(h,hbMonthCells(bfMonth),bfSel,'bfTap');
 const vw=document.getElementById('bfValWrap'),on=h.type==='pos'&&hbHasVal(h);
 vw.classList.toggle('hidden',!on);
 if(on){if(bfVk!==h.id)bfValInit(h);bfValRows(h)}
 bfN();openSheet('sheetHabitBackfill')}
function bfSave(){const h=bfHabit();if(!h||!bfSel.length){toast('先选要补记的日期');return}
 const st=document.getElementById('bfStreak').classList.contains('on');
 const useVal=h.type==='pos'&&hbHasVal(h);
 const val=useVal?(h.group?bfV.map(v=>+v||0):[+bfV[0]||0]):null;
 const old=bfSel.map(b=>hbDayGet(h,b).ok?0:1).reduce((a,b)=>a+b,0);
 bfSel.slice().sort((a,b)=>a-b).forEach(b=>hbDaySet(h,b,1,useVal?val:null));
 h.mDone=(h.mDone||0)+old;if(st)h.streak=(h.streak||0)+old;
 closeSheet('sheetHabitBackfill');renderHabits();renderHabitStat();if(curPage==='p-habitDetail')renderHabitDetail();
 const vTxt=useVal?(h.group?' · 子项 '+bfV.join('/'):' · 数值 '+(+bfV[0]||0)):'';
 toast('已为「'+h.n+'」补记 '+bfSel.length+' 天'+vTxt+(st?' · 已计入连续':' · 未计入连续'));
 bfSel=[]}
