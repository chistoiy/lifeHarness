function bxd(t) { return String(t == null ? '' : t).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;') }
function bxDot(s) { return '<span class="bx-dot bx-hs-' + s.hs + '"></span>' }
function bxKind(s) { return '<span class="bx-k ' + (s.kind === 'rule' ? '' : s.kind) + '">' + BX_KINDN[s.kind] + '</span>' }
function bxSel(s) { return !!s && s.on === 1 && (s.kind === 'rule' || s.kind === 'part') && s.hs !== 'dead' }
function bxUsable(s) { return s.kind === 'js' ? '需 JS 引擎 · 当前构建未启用' : s.kind === 'web' ? '需浏览器渲染 · 本期未启用' : s.hs === 'dead' ? '源已失效' : s.hs === 'fuse' ? '已熔断 · 本轮跳过' : '' }

function bxBad(s) { return s.hs === 'dead' || s.hs === 'fuse' }

function bxAddParse() {

  BXAI.json = (document.getElementById('bxaiJson') || {}).value || '';
  const rows = [];
  BXAI.urls.split(/[\n,，\s]+/).filter(Boolean).forEach(function (u) {
    const dup = BX_SUBS.find(x => x.url === u);
    if (/^https?:\/\//.test(u)) {
      const sub = /shuyuan|booksource|bookSource|legado|\/api\//.test(u) || /\.json$|\.txt$/.test(u);
      rows.push({ t: sub ? '订阅' : '单源', src: u, n: sub ? (dup ? dup.n : 22) : 1, dup: dup ? '同名订阅已存在 · 合并更新' : '', note: sub ? '整份源集合 · 只取规则不取正文' : '按 bookUrlPattern 建单源', ok: 1 });
    } else rows.push({ t: '无效', src: u, n: 0, dup: '不是可识别的地址', ok: 0 });
  });
  BXAI.files.forEach(function (nm) {
    const f = BXAI_FILES.find(x => x.nm === nm) || { kind: '源数组', n: 1, note: '自定义文件' };
    const dup = f.kind === '订阅' ? BX_SUBS.some(x => x.url.indexOf(nm) >= 0) : false;
    rows.push({ t: f.kind, src: nm, n: f.n, dup: dup ? '与已有订阅重复 · 跳过' : '', note: f.note, ok: 1, file: 1 });
  });
  if (BXAI.json.trim()) {
    try {
      const j = JSON.parse(BXAI.json);
      const arr = Array.isArray(j) ? j : [j];
      const good = arr.filter(x => x && (x.bookSourceUrl || x.bookSourceName));
      const bad = arr.length - good.length;
      if (!good.length) rows.push({ t: '无效', src: '粘贴内容 ' + arr.length + ' 条', n: 0, dup: '缺 bookSourceUrl / bookSourceName 字段', ok: 0 });
      else rows.push({ t: arr.length > 1 ? '源数组' : '单源', src: '粘贴 JSON', n: good.length, dup: '', note: bad ? bad + ' 条缺 bookSourceUrl 已丢弃' : '字段完整 · 可直接建源', ok: 1 });
    } catch (e) { rows.push({ t: '无效', src: '粘贴 JSON', n: 0, dup: '解析失败：' + String(e.message).slice(0, 28), ok: 0 }) }
  }
  if (!rows.length) { document.getElementById('bxaiPrev').innerHTML = '<div class="bx-empty">没有可解析的内容 · 请填链接、选文件或粘贴 JSON</div>'; BXAI.rows = null; return }
  const news = rows.filter(r => r.ok && !r.dup);

function bxAddImport() {

/* ---------- 逐章抓取引擎（防盗链 / 失效 / 超时 → 按优先级自动换源） ---------- */
function bxTrySrc(s, idx) {
  if (!bxSel(s)) return { ok: 0, why: bxUsable(s) || '未启用' };
  if (s.kind === 'js') return { ok: 0, why: '规则含 @js · JS 引擎未启用' };
  if (s.kind === 'web') return { ok: 0, why: '403 防盗链 · 需浏览器渲染挑战页' };
  if (s.hs === 'dead') return { ok: 0, why: '站点 404 · 源已失效' };
  if (s.hs === 'fuse') return { ok: 0, why: '连续 ' + s.fails + ' 次超时 · 本轮熔断' };
  if (s.kind === 'part' && idx % 7 === 3) return { ok: 0, why: '正文规则缺 textNodes · 该章取到空段' };
  return { ok: 1, ms: s.ms || 700 };
}
function bxFetch(b, idx, quiet) {
  const was = b.bsrc.srcId;
  const order = BS_SET.bsAuto
    ? [was].concat(BX_ORDER.filter(x => x !== was)).concat(BX_SRC.map(s => s.id).filter(id => id !== was && BX_ORDER.indexOf(id) < 0))
    : [was];
  const tried = [];
  for (const id of order) {
    const s = bxs(id); if (!s) continue;
    const r = bxTrySrc(s, idx);
    if (r.ok) {
      b.bsrc = { srcId: s.id, srcNm: s.nm, total: b.bsrc.total, cached: Math.max(b.bsrc.cached, idx), dead: false };
      if (id !== was) b.src = '书源·' + s.nm;
      if (b.epub) { b.epub.size = (b.bsrc.cached * 12) + ' KB'; b.epub.path = '书源缓存 · ' + s.nm; }
      if (!quiet) toast('第 ' + idx + ' 章 · ' + s.nm + ' ' + r.ms + 'ms' + (id !== was && BS_SET.bsAuto ? ' · 已自动换源' : ''));
      return { ok: 1, s: s, changed: id !== was };
    }
    tried.push(s.nm + '：' + r.why);
  }
  b.bsrc.dead = true;
  return { ok: 0, tried: tried };
}

function bxTaskAll() {
  const busy = BX_TASKS.some(t => t.st === 'run');
  BX_TASKS.forEach(t => {
    if (busy && t.st === 'run') { t.st = 'pause'; t.err = '停在第 ' + (t.done + 1) + ' 章 · manifest 已落盘' }
    else if (!busy && t.st === 'pause') { t.st = 'run'; t.err = '' }
  });
  clearInterval(BXTIMER);
  if (!busy && BX_TASKS.some(t => t.st === 'run')) bxTaskRun(BX_TASKS.find(t => t.st === 'run'));
  bxTaskRender(); toast(busy ? '已全部暂停 · manifest 已落盘' : '已全部继续');

  t.done = 0; t.st = 'pause'; t.mb = '—'; bxTaskRender(); toast('已释放《' + t.bk + '》正文缓存 · 记录与书摘保留');

  toast('重取目录：' + t.total + ' → ' + (t.total + t.upd) + ' 章 · 只抓新增');

let BXTIMER = null;
function bxTaskRun(t, only) {
  clearInterval(BXTIMER); bxTaskRender();
  const w = bxTaskWin(t), target = w.b, step = 0;
  BXTIMER = setInterval(() => {
    if (!t || t.st !== 'run') { clearInterval(BXTIMER); return }
    t.done++;
    if (!t.range && t.src === 'r2' && t.done === 10) { t.st = 'fail'; t.retry = BS_SET.bsRetry; t.err = '第 ' + t.done + ' 章 · 正文规则未取到段落（缺 textNodes）'; clearInterval(BXTIMER); bxTaskRender(); return }
    const b = bxBk(t.bk); if (b && b.bsrc) b.bsrc.cached = Math.max(b.bsrc.cached, t.done);
    if (t.done >= target) {
      const got = t.range ? (t.done - t.range[0] + 1) : t.total;
      t.st = 'done'; t.mb = (got * 12 / 1024).toFixed(1) + ' MB'; t.at = '刚刚'; t.err = ''; if (!t.range) t.upd = 0;
      if (b && !t.range) { b.epub.st = 'rdy'; b.epub.size = t.mb; }
      try { renderShelf() } catch (e) { }
      clearInterval(BXTIMER); bxTaskRender();
      toast('《' + t.bk + '》' + (t.range ? '第 ' + t.range[0] + '–' + t.range[1] + ' 章' : only ? '新增 ' + only + ' 章' : '全本') + '已入段落快照 · 缓存 ' + (b && b.bsrc ? b.bsrc.cached : target) + '/' + t.total + ' 章'); return;
    }
    const bar = document.getElementById('bxTaskRoot'); if (bar) { const row = bar.querySelectorAll('.bx-prog>i')[BX_TASKS.indexOf(t)]; if (row) row.style.width = Math.round(Math.min(100, (t.range ? (t.done - t.range[0] + 1) / (t.range[1] - t.range[0] + 1) : t.done / t.total) * 100)) + '%'; }
    if (t.done % 24 === 0) bxTaskRender();
    void step;
  }, 380);
}
