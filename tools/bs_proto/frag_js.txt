function bxs(id) { return BX_SRC.find(s => s.id === id) || null }
function bxd(t) { return String(t == null ? '' : t).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;') }
function bxDot(s) { return '<span class="bx-dot bx-hs-' + s.hs + '"></span>' }
function bxKind(s) { return '<span class="bx-k ' + (s.kind === 'rule' ? '' : s.kind) + '">' + BX_KINDN[s.kind] + '</span>' }
function bxSel(s) { return !!s && s.on === 1 && (s.kind === 'rule' || s.kind === 'part') && s.hs !== 'dead' }
function bxUsable(s) { return s.kind === 'js' ? '需 JS 引擎 · 当前构建未启用' : s.kind === 'web' ? '需浏览器渲染 · 本期未启用' : s.hs === 'dead' ? '源已失效' : s.hs === 'fuse' ? '已熔断 · 本轮跳过' : '' }
function bxProto(nm) { return BX_PROTO.find(p => p.nm === nm) || BX_PROTO[0] }
function bxBad(s) { return s.hs === 'dead' || s.hs === 'fuse' }

function bxRegBook(srcId, nm) {
  const s = bxs(srcId) || BX_SRC[0];
  const t = (nm || (BXV.b && BXV.b.t) || '剑来').trim();
  const on = bxProto(t);
  let b = bxBk(t);
  if (b) {
    b.online = 1; b.local = b.local || 1;
    b.bsrc = { srcId: s.id, srcNm: s.nm, total: b.bsrc ? b.bsrc.total : on.total, cached: b.bsrc ? b.bsrc.cached : on.cached, dead: bxBad(s) };
    if (!b.tags.includes('在线')) b.tags.push('在线');
    if (!b.epub) b.epub = { fmt: '书源正文', size: (b.bsrc.cached * 12) + ' KB', path: '书源缓存 · ' + s.nm, st: 'rdy' };
    bxTocOf(b);
    try { renderShelf() } catch (e) { }
    return { b: b, dup: true };
  }
  const id = 'bx_' + t.replace(/[^0-9A-Za-z\u4e00-\u9fa5]/g, '').slice(0, 8) + '_' + (Date.now() % 10000);
  const pg = +((BXV.b && BXV.b.wc) || on.wc || '').replace(/[^0-9]/g, '') || on.pg;
  b = { id: id, n: t, a: (BXV.b && BXV.b.a) || on.a, st: '在读', p: on.p, pg: Math.max(60, Math.round(pg / 10000)), rp: 0, q: 0, tags: ['在线'], c: on.c, rate: 0, isbn: '—', pub: '网络连载', yr: 2026, src: '书源·' + s.nm, start: '09-29', end: '', min: 0, times: 0, note: '', fin: '', local: 1, online: 1, epub: { fmt: '书源正文', size: (on.cached * 12) + ' KB', path: '书源缓存 · ' + s.nm, st: 'rdy' }, bsrc: { srcId: s.id, srcNm: s.nm, total: on.total, cached: on.cached, dead: bxBad(s) } };
  b.rp = Math.round(b.p / 100 * b.pg);
  BS_BOOKS.push(b); bsRegister(b);

  try { renderShelf() } catch (e) { }
  return { b: b, dup: false };
}

let BXTIMER = null;
function bxTaskRun(t, only) {
  clearInterval(BXTIMER); bxTaskRender();
  const target = t.total, step = 0;
  BXTIMER = setInterval(() => {
    if (!t || t.st !== 'run') { clearInterval(BXTIMER); return }
    t.done++;
    if (t.src === 'r2' && t.done === 10) { t.st = 'fail'; t.retry = BS_SET.bsRetry; t.err = '第 ' + t.done + ' 章 · 正文规则未取到段落（缺 textNodes）'; clearInterval(BXTIMER); bxTaskRender(); return }
    const b = bxBk(t.bk); if (b && b.bsrc) b.bsrc.cached = Math.max(b.bsrc.cached, t.done);
    if (t.done >= target) {
      t.st = 'done'; t.mb = (target * 12 / 1024).toFixed(1) + ' MB'; t.at = '刚刚'; t.err = ''; t.upd = 0;
      if (b) { b.epub.st = 'rdy'; b.epub.size = t.mb; }
      clearInterval(BXTIMER); bxTaskRender(); toast('《' + t.bk + '》' + (only ? '新增 ' + only + ' 章' : '全本') + '已入段落快照 · ' + target + ' 章'); return;
    }
    const bar = document.getElementById('bxTaskRoot'); if (bar) { const row = bar.querySelectorAll('.bx-prog>i')[BX_TASKS.indexOf(t)]; if (row) row.style.width = Math.round(t.done / t.total * 100) + '%'; }
    if (t.done % 24 === 0) bxTaskRender();
    void step;
  }, 380);
}
