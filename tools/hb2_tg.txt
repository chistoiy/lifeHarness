function hbTrigOpen(i){tgI=i;const h=HABITS[i];tgR='';
 document.getElementById('tgTitle').textContent='记一次触发 · '+h.n;
 document.getElementById('tgSub').textContent='如实记录即可 · 只统计，不评判';
 document.getElementById('tgReasons').innerHTML=HB_REASONS.map(r=>`<button class="chip !text-[10px]" onclick="tgPick('${r}',this)">${r}</button>`).join('');
 document.getElementById('tgAmtUnit').textContent=h.unit||'次';
 document.getElementById('tgAmtRow').classList.toggle('hidden',!h.unit);
 document.getElementById('tgAmt').value=h.goal||1;
 document.getElementById('tgNote').value='';openSheet('sheetHabitTrigger')}
function tgPick(r,el){tgR=r;document.querySelectorAll('#tgReasons .chip').forEach(b=>b.classList.remove('chip-on'));el.classList.add('chip-on')}
function tgSave(){const h=HABITS[tgI],note=document.getElementById('tgNote').value.trim();
 const amt=Math.max(0,+document.getElementById('tgAmt').value||0);
 h.w[6]=0;h.triggers=(h.triggers||0)+1;h.mAvoid=Math.max(0,(h.mAvoid||0)-1);h.streak=0;
 h.v=h.v||[0,0,0,0,0,0,0];h.v[6]=(h.v[6]||0)+amt;
 h.slips=h.slips||[];h.slips.unshift({dt:'09-19',rs:tgR||'未填情境',note:note,amt:amt+' '+(h.unit||'次')});
 closeSheet('sheetHabitTrigger');renderHabits();if(curPage==='p-habitDetail')renderHabitDetail();
 toast('已记录 · 这次 '+amt+' '+(h.unit||'次')+' · 本周 '+(h.triggers)+'/'+h.target+(hbNegLeft(h)?' · 还剩 '+hbNegLeft(h)+' 次额度':' · 本周额度已用满'))}
